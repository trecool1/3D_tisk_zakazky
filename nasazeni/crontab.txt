# Kanban zakázek Cadmia3D — cron pro uživatele cadmia
#   crontab ~/kanban-stage/nasazeni/crontab.txt      (nahradí celý crontab)
#   nebo: crontab -e  a vložit řádky níže
# Běží jako cadmia, protože PHP-FPM pool kanbanu jede taky jako cadmia —
# všichni zapisovatelé do kanban.sqlite jsou pak stejný uživatel.
#
# ── REŽIM PROVOZU ────────────────────────────────────────────────────────────
# E-mail je vždy primární: kalkulátor (order.php) posílá KAŽDOU poptávku na
# pokusss74@seznam.cz (text + PDF nabídka + JSON + modely). Funguje to bez
# ohledu na kanban — když kanban vypneš, o žádnou poptávku nepřijdeš.
#
# Kanban jede vedle toho v režimu PULL: čte poptavky/objednavky.json kalkulátoru
# a zakládá karty. Vypnout kanban = zakomentovat řádek import-kalkulator.php.
# `--bez-mailu` = kanban při importu NEROZESÍLÁ vlastní potvrzení/upozornění,
# aby nechodily dvakrát (jednou z kalkulátoru, podruhé z kanbanu).
# Kalkulátor na PUSH nepřepínat ($WEBHOOK v order.php nechat prázdný), ať
# e-mailová cesta zůstane jako záloha.
# ────────────────────────────────────────────────────────────────────────────

# PULL poptávek z poptavky/objednavky.json kalkulátoru (tiše, bez e-mailů z kanbanu)
*/1 * * * * php /var/www/kanban-app/cron/import-kalkulator.php --bez-mailu >> /var/www/kanban-data/cron.log 2>&1

# ranní přehled dílně: co tisknout, co expedovat, po termínu, neodpovězené nabídky
10 7 * * *   php /var/www/kanban-app/cron/denni.php             >> /var/www/kanban-data/cron.log 2>&1

# noční záloha DB (VACUUM INTO) + tar modelů, drží se 14 dní
30 2 * * *   php /var/www/kanban-app/cron/zaloha.php            >> /var/www/kanban-data/cron.log 2>&1

# příchozí pošta z IMAP — párování odpovědí zákazníků do konverzace, zbytek do
# Nezařazeno. Schránka pokusss74@seznam.cz je sdílená s upozorněními z kalkulátoru:
# ta jdou podle hlavičky X-Poptavka-Cislo potichu pryč, cron ale VŠECHNU poštu
# ve schránce označuje jako přečtenou. Ideálně samostatná schránka jen pro zakázky.
*/1 * * * *  php /var/www/kanban-app/cron/posta.php             >> /var/www/kanban-data/cron.log 2>&1
